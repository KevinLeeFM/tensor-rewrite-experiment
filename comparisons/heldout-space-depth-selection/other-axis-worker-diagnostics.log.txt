theorem Heldout.OtherAxisIntegration.full_index : ActualFullIndexGoal.Goal :=
fun {rank outerRank} {shape} {inner} {outer} {α} plan input out =>
  funext fun i =>
    if hc : i = concatAxis plan.axes then
      Eq.symm hc ▸ Fin.ext (OriginalCoordinateIntegration.moved_axis_equality plan input out)
    else
      if hs : i = plan.axes.split then
        Eq.ndrec (motive := fun i =>
          ¬i = concatAxis plan.axes → originalIndex plan out i = replacementIndex plan input out i)
          (fun hc => RowMajorSplitIntegration.split plan input out) (Eq.symm hs) hc
      else OtherAxisProof.other plan input out i hc hs
theorem Heldout.OtherAxisIntegration.rewrite : Goal :=
fun rank outerRank shape inner outer plan α input =>
  funext fun out => read_equal_of_index_equal plan input out (full_index plan input out)
