theorem moved_axis {rank outerRank : Nat} {shape : Shape rank} {inner : Shape (rank + 1)}
    {outer : Shape outerRank} {α : Type} (plan : Conditions shape inner outer)
    (input : Tensor shape α) (out : Index outer) :
    (originalIndex plan out (concatAxis plan.axes)).val =
      (replacementIndex plan input out (concatAxis plan.axes)).val := by
  have htransport := Goal_proved plan input out
  simp [IndexTransportAttempt.originalIndex, IndexTransportAttempt.replacementIndex,
    Heldout.ChunkIntervals.chunkBlocks_total, Heldout.SpaceDepthCandidate.chunk_count]

    (originalIndex plan out plan.axes.split).val =
      (replacementIndex plan input out plan.axes.split).val := by
  have htransport := Goal_proved plan input out
  simp [IndexTransportAttempt.originalIndex, IndexTransportAttempt.replacementIndex,
    Heldout.ChunkIntervals.chunkBlocks_total, Heldout.SpaceDepthCandidate.chunk_count]
